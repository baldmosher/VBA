''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''
' modKeyState
' By Chip Pearson, www.cpearson.com, chip@cpearson.com
' This module contains functions for testing the state of the SHIFT, ALT, and CTRL
' keys.
''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''

        
        ''''''''''''''''''''''''''''''''''''''''''''''''''''
        ' Declaration of GetKeyState API function. This
        ' tests the state of a specified key.
        ''''''''''''''''''''''''''''''''''''''''''''''''''''
        Private Declare PtrSafe Function GetKeyState Lib "user32" ( _
            ByVal nVirtKey As Long) As Integer
        'NB: amended to include PtrSafe according to https://msdn.microsoft.com/en-us/library/ee691831.aspx?f=255&MSPPError=-2147217396#odc_office2010_Compatibility32bit64bit_ApplicationProgrammingInterfaceCompatibility
            
        ''''''''''''''''''''''''''''''''''''''''''
        ' This constant is used in a bit-wise AND
        ' operation with the result of GetKeyState
        ' to determine if the specified key is
        ' down.
        ''''''''''''''''''''''''''''''''''''''''''
        Private Const KEY_MASK As Integer = &HFF80 ' decimal -128
        
        '''''''''''''''''''''''''''''''''''''''''
        ' KEY CONSTANTS. Values taken
        ' from VC++ 6.0 WinUser.h file.
        '''''''''''''''''''''''''''''''''''''''''
        Private Const VK_LSHIFT = &HA0
        Private Const VK_RSHIFT = &HA1
        Private Const VK_LCONTROL = &HA2
        Private Const VK_RCONTROL = &HA3
        Private Const VK_LMENU = &HA4
        Private Const VK_RMENU = &HA5
        '''''''''''''''''''''''''''''''''''''''''
        ' The following four constants simply
        ' provide other names, CTRL and ALT,
        ' for CONTROL and MENU. "CTRL" and
        ' "ALT" are more familiar than
        ' "CONTROL" and "MENU". These constants
        ' provide no additional functionality.
        ' They simply provide more familiar
        ' names.
        '''''''''''''''''''''''''''''''''''''''''
        Private Const VK_LALT = VK_LMENU
        Private Const VK_RALT = VK_RMENU
        Private Const VK_LCTRL = VK_LCONTROL
        Private Const VK_RCTRL = VK_RCONTROL
        
         ''''''''''''''''''''''''''''''''''''''''''''
        ' The following constants are used to specify,
        ' when testing CTRL, ALT, or SHIFT, whether
        ' the Left key, the Right key, either the
        ' Left OR Right, or BOTH the Left AND Right
        ' key is down.
        '
        ' By default, the key-test procedures make
        ' no distinction between the Left and Right
        ' keys.
        ''''''''''''''''''''''''''''''''''''''''''''
        Private Const BothLeftAndRightKeys = 0   ' Note: Bit-wise AND of LeftKey and RightKey
        Private Const LeftKey = 1
        Private Const RightKey = 2
        Private Const LeftKeyOrRightKey = 3      ' Note: Bit-wise OR of LeftKey and RightKey
        
        
        Private Function IsShiftKeyDown(Optional LeftOrRightKey As Long = LeftKeyOrRightKey)
        ''''''''''''''''''''''''''''''''''''''''''''''''
        ' IsShiftKeyDown
        ' Returns TRUE or FALSE indicating whether the
        ' SHIFT key is down.
        '
        ' If LeftOrRightKey is omitted or LeftKeyOrRightKey,
        ' the function return TRUE if either the left or the
        ' right SHIFT key is down. If LeftKeyOrRightKey is
        ' LeftKey, then only the Left SHIFT key is tested.
        ' If LeftKeyOrRightKey is RightKey, only the Right
        ' SHIFT key is tested. If LeftOrRightKey is
        ' BothLeftAndRightKeys, the codes tests whether
        ' both the Left and Right keys are down. The default
        ' is to test for either Left or Right, making no
        ' distiction between Left and Right.
        ''''''''''''''''''''''''''''''''''''''''''''''''
            Dim Res As Long
            
            Select Case LeftOrRightKey
                Case LeftKey
                    Res = GetKeyState(VK_LSHIFT) And KEY_MASK
                Case RightKey
                    Res = GetKeyState(VK_RSHIFT) And KEY_MASK
                Case BothLeftAndRightKeys
                    Res = (GetKeyState(VK_LSHIFT) And GetKeyState(VK_RSHIFT) And KEY_MASK)
                Case Else
                    Res = GetKeyState(vbKeyShift) And KEY_MASK
            End Select
            
            IsShiftKeyDown = CBool(Res)
        End Function

Sub xlU_EmptyFolder(ByVal fdr As String, Optional ByVal AlsoRmDir As Boolean _
    , Optional ByVal DoMsgs As Boolean)
'v7.03 2013-05-22 12:22
'delete all files from specified folder [and remove folder]

Const pp As String = "WARNING: cannot undo this action"
Const tt As String = "xlU_EmptyFolder"
If MsgBox(pp, vbOKCancel Or vbCritical, tt) = vbCancel Then Exit Sub

Const cBsl As String = "\"
Const cFsl As String = "/"
Const cSdS As String = "*.*"
Const emsg As String = "xlU_EmptyFolder failed: path not valid, no slashes"

Dim fn As String, sl As String
If InStr(fdr, cBsl) > 0 Then sl = cBsl
If InStr(fdr, cFsl) > 0 Then sl = cFsl
If sl = vbNullString Then
    If DoMsgs = True Then MsgBox emsg, vbCritical, tt
    Exit Sub
End If

If Right(fdr, 1) <> sl Then fdr = fdr & sl

fn = Dir(fdr & cSdS)
Do While fn <> vbNullString
On Error Resume Next
    Kill fdr & fn
On Error GoTo 0
    fn = Dir
Loop  'all files in folder

If AlsoRmDir = True Then RmDir fdr

End Sub

Function xlU_Clean_Special(ByVal str As String, Optional ByVal CropLength As Boolean = True _
    , Optional ByVal OnlyFilename As Boolean _
    , Optional ByVal OnlyVBObjectName As Boolean _
    , Optional ByVal OnlySheetName As Boolean) As String
'v33.06 2017-06-16 14:36 - added OnlySheetName option and removes []
'v9.00 2013-07-25 19:31
'removes invalid special characters from path/file/VBObject name string
' CropLength:=True stops message box warnings and autocrops string to 128 chars
' OnlyFilename:=True also removes slashes \ /
' OnlyVBObjectName:=True also removes slashes, spaces, hyphens, commas, periods
' OnlySheetName:=True also crops to 31 characters

'constants (commented if defined below)
'Const scUS As String = "_"
Const VBNameLength = 35
Const SheetNameLength = 31
Const SharePointFileNameLength = 128
'consistency checks are done below for the following constants
Const uniMin As Byte = 0    'first universally forbidden character, MUST BE ZERO
Const uniMax As Byte = 15   'last universally forbidden character, number set according to how many forbidden characters
Const slaFSL As Byte = 16   'first slash, always Fsl, AWAYS uniMax+1
Const slaBSL As Byte = 17   'last slash, always Bsl, AWAYS slaFSL+1
Const vboMin As Byte = 18   'first VBA forbidden character, AWAYS slaBSL+1
Const vboMax As Byte = 24   'last VBA forbidden character, number set according to how many forbidden characters

Dim b As Integer, c As Integer, pp As String
Const tt As String = "ERROR in xlU_Clean_Special"
Dim sc(uniMin To vboMax) As String
sc(uniMin) = "~"     'unimin referenced specifically below
sc(uniMin + 1) = Chr(34) 'Chr(34) = " (quotemark)
sc(uniMin + 2) = "#"
sc(uniMin + 3) = "%"
sc(uniMin + 4) = "&"
sc(uniMin + 5) = "*"
sc(uniMin + 6) = ":"
sc(uniMin + 7) = "<"
sc(uniMin + 8) = ">"
sc(uniMin + 9) = "?"
sc(uniMin + 10) = "{"
sc(uniMin + 11) = "|"
sc(uniMin + 12) = "}"
sc(uniMin + 13) = "["
sc(uniMin + 14) = "]"
If uniMin + 15 <> uniMax Then   'consistency check
    pp = "uniMin + 15 <> uniMax"
    MsgBox pp, vbCritical, tt
    End
End If
sc(uniMax) = ".."
'slashes for filenames and VB Object names (NOT paths)
sc(slaFSL) = "/"
sc(slaBSL) = "\"
'hyphen & space & comma & period & brackets & apostrophe for VB Object names
sc(vboMin) = "-"
sc(vboMin + 1) = " "
sc(vboMin + 2) = ","
sc(vboMin + 3) = "."
sc(vboMin + 4) = "("
sc(vboMin + 5) = "'"
If vboMin + 6 <> vboMax Then    'consistency check
    pp = "vboMin + 6 <> vboMax"
    MsgBox pp, vbCritical, tt
    End
End If
sc(vboMax) = ")"

'remove special characters from all
For b = uniMin To uniMax
    str = Replace(str, sc(b), vbNullString)
Next b

'check filename length (length AFTER the LAST slash max 128 chars)
b = InStr(1, str, sc(slaFSL))  'look for fwd slash
If b > 0 Then
    str = Replace(str, sc(slaBSL), sc(slaFSL))  'remove all back slashes
    Do Until b = 0  'until last slash found
        c = b       'c is position of last slash
        b = b + 1                   'next position
        b = InStr(b, str, sc(slaFSL))   'next position
    Loop
Else  'no fwd slashes
    b = InStr(1, str, sc(slaBSL))  'look for back slash
    If b > 0 Then
        str = Replace(str, sc(slaFSL), sc(slaBSL))  'remove all fwd slashes
        Do Until b = 0  'until last slash found
            c = b       'c is position of last slash
            b = b + 1                   'next position
            b = InStr(b, str, sc(slaBSL))   'next position
        Loop
    End If
End If
'c is position of last slash, or 0 if no slashes
If Len(str) - c > SharePointFileNameLength Then
    If CropLength = True Then
        str = Left(str, VBNameLength)
    Else
        pp = "WARNING: filename > " & SharePointFileNameLength & " chars" & vbLf & vbLf & str
        MsgBox pp, vbCritical, tt
        End
    End If
End If

If OnlyFilename = True Or OnlyVBObjectName = True Or OnlySheetName = True Then
'swap all slashes for spaces - NOT suitable for paths!
    For b = slaFSL To slaBSL
        c = InStr(str, sc(b))
        Do While c > 0
            str = Left(str, c - 1) & Replace(Right(str, Len(str) - c), sc(b), cSpc)
            c = InStr(str, sc(b))
        Loop
    Next b
End If

If OnlyVBObjectName = True Then
'swap hyphens & spaces & periods for underscore in VB object name
    Const scUS As String = "_"
    For b = slaFSL To vboMax
        str = Replace(str, sc(b), scUS)
    Next b
'then remove invalid characters from start of string
    Dim c1 As String
    c1 = Left(str, 1)
    Do While c1 = scUS Or c1 = sc(18) Or IsNumeric(c1)
        str = Right(str, Len(str) - 1)
        c1 = Left(str, 1)
    Loop
'remove double underscore
    Do While InStr(str, scUS & scUS) > 0
        str = Replace(str, scUS & scUS, scUS)
    Loop
'remove needless underscores from end of string
    Do While Right(str, 1) = scUS
        str = Left(str, Len(str) - 1)
    Loop
    'check object name length
    If Len(str) > VBNameLength Then
        If CropLength = True Then
            str = Left(str, VBNameLength)
        Else
            pp = "WARNING: object name > " & VBNameLength & " chars"
            MsgBox pp, vbCritical, tt
        End If
    End If
End If

'Requires a reference Lookup file to be in place in order to create structure.  This lookup must
'have the specified named column headers, but can be in any order.  The macro only uses
'columns that are specified.  The contents of these columns determines the folder names.

'syntax:
'   xlU_Deploy_Folders "C:\Users\" & Environ("Username") & "\Documents\recursetest\"

'e.g.
'   AddAlphanumericSubfolders = True   will add A, B, C, D, E, ... Z and "0-9" subfolders within each Level1 folder
'   sLevel1Header = "Country Code"     mandatory top level (within this, can optionally AddAlphanumericSubfolders)
'   sLevel2Header = "Sub-Region"       level 1 grouped by this, if not required use ""
'   sLevel3Header = "Region Code"      level 2 grouped by this, if not required use ""

'NB:
'   Gaps (empty cells) in Level1 column will stop the macro partway down the list.  Use
'   apostrophe (') for all blank cells in this column.  Zero-length strings ("") will be
'   replaced by sNotAssigned
Const sNotAssigned As String = "_other"

On Error Resume Next

'check BasePath exists
If Right(BasePath, 1) <> "\" Then BasePath = BasePath & "\"
If Dir(BasePath, vbDirectory) <> "." Then  'error here means folder not found
    MkDir BasePath
    If Dir(BasePath, vbDirectory) <> "." Then  'error here means folder not created
        MsgBox "BasePath not found" & vbLf & vbLf & BasePath, vbCritical, "Fatal Error"
        Exit Sub
    End If
End If

'check LevelLookup exists
If Dir(LevelLookup, vbDirectory) = "" Then  'error here means file not found
    MsgBox "LevelLookup not found" & vbLf & vbLf & LevelLookup, vbCritical, "Fatal Error"
    Exit Sub
End If

'open LevelLookup, check that all required column headers exist in row 1
Dim wbLU As Excel.Workbook, wsLU As Excel.Worksheet, rLUhdr As Excel.Range
Dim bColumnFail As Boolean, cmax As Byte, rmax As Long
Dim bLevel1 As Byte, bLevel2 As Byte, bLevel3 As Byte
Dim rLevel1 As Excel.Range, rLevel2 As Excel.Range, rLevel3 As Excel.Range
Set wbLU = Workbooks.Open(LevelLookup, False, True)
If wbLU Is Nothing Then
    MsgBox "LevelLookup not found" & vbLf & vbLf & LevelLookup, vbCritical, "Fatal Error"
    Exit Sub
Else
    Set wsLU = wbLU.Sheets(1)
    Set rLUhdr = wsLU.Rows(1)
    bLevel1 = xl.Match(sLevel1Header, rLUhdr, 0)
    bLevel2 = xl.Match(sLevel2Header, rLUhdr, 0)
    bLevel3 = xl.Match(sLevel3Header, rLUhdr, 0)
    If bLevel1 = 0 And sLevel1Header <> "" Then
        bColumnFail = True
    Else
        cmax = 1
        Set rLevel1 = rLUhdr.Cells(bLevel1)
        Set rLevel1 = wsLU.Range(rLevel1.Offset(1), rLevel1.End(xlDown))
        rmax = rLevel1.Cells.Count
        If bLevel2 = 0 And sLevel2Header <> "" Then
            bColumnFail = True
        Else
            cmax = 2
            Set rLevel2 = rLUhdr.Cells(bLevel2)
            Set rLevel2 = wsLU.Range(rLevel2.Offset(1), rLevel2.End(xlDown))
            If bLevel3 = 0 And sLevel3Header <> "" Then
                bColumnFail = True
            Else
                cmax = 3
                Set rLevel3 = rLUhdr.Cells(bLevel3)
                Set rLevel3 = wsLU.Range(rLevel3.Offset(1), rLevel3.End(xlDown))
            End If
        End If
    End If
    If bColumnFail Then
        MsgBox "LevelLookup does not contain correct columns for chosen options.  Please check the macro in VBA for advice.  (NB: LevelLookup was opened Read-Only)", vbCritical, "Fatal Error"
        Exit Sub
    End If
End If

'load Level1, Level2, Level3 into array (memory) prior to recurse
Dim arrRecurse() As String, NewPath As String, r As Integer, c As Byte, f As Long, a As Long
If MsgBox("LevelLookup contains " & rmax & " " & sLevel1Header & "s.  Continue?", vbYesNo Or vbQuestion, "Fatal Error") = vbNo Then
    Exit Sub
End If
ReDim arrRecurse(1 To rmax, 0 To cmax) As String
'so, e.g. for countries:
'AD is always arrRecurse(1,1), EMEA is always arrRecurse(1,2), EURO always arrRecurse(1,3)
For r = 1 To rmax
    arrRecurse(r, 1) = rLevel1.Cells(r)
    If arrRecurse(r, 1) = "" Then arrRecurse(r, 1) = sNotAssigned
    If cmax >= 2 Then
        arrRecurse(r, 2) = rLevel2.Cells(r)
        If arrRecurse(r, 2) = "" Then arrRecurse(r, 2) = sNotAssigned
        If cmax = 3 Then
            arrRecurse(r, 3) = rLevel3.Cells(r)
            If arrRecurse(r, 3) = "" Then arrRecurse(r, 3) = sNotAssigned
        End If
    End If
    If DebugMode Then Debug.Print arrRecurse(r, 1) & "  " & arrRecurse(r, 2) & "  " & arrRecurse(r, 3)
Next r

'close lookup
wbLU.Close False

'now recurse through array, creating folders
For r = 1 To rmax
    NewPath = BasePath
    If DebugMode Then Debug.Print NewPath
    If cmax = 3 Then
        NewPath = NewPath & arrRecurse(r, 3) & "\"
        If DebugMode Then Debug.Print NewPath
        If Dir(NewPath, vbDirectory) <> "." Then
            MkDir NewPath
            f = f + 1
        Else
            a = a + 1
        End If
    End If
    If cmax >= 2 Then
        NewPath = NewPath & arrRecurse(r, 2) & "\"
        If DebugMode Then Debug.Print NewPath
        If Dir(NewPath, vbDirectory) <> "." Then
            MkDir NewPath
            f = f + 1
        Else
            a = a + 1
        End If
    End If
    If cmax >= 1 Then
        NewPath = NewPath & arrRecurse(r, 1) & "\"
        If DebugMode Then Debug.Print NewPath
        If Dir(NewPath, vbDirectory) <> "." Then
            MkDir NewPath
            f = f + 1
        Else
            a = a + 1
        End If
    End If
    If AddAlphanumericSubfolders Then
        If Dir(NewPath & "0-9\", vbDirectory) <> "." Then
            MkDir NewPath & "0-9\"
            f = f + 1
        Else
            a = a + 1
        End If
        For c = 65 To 90
            If Dir(NewPath & Chr(c) & "\", vbDirectory) <> "." Then
                MkDir NewPath & Chr(c) & "\"
                f = f + 1
            Else
                a = a + 1
            End If
        Next c
    End If
Next r

MsgBox "Done, created " & f & " new folders, " & a & " already existed"

End Sub

Function xlU_Text_Capitalise_Words(ByVal s As String) As String
'v26.00 2014-09-12 14:02 - added specific exceptions listed below
'v25.04 2014-09-12 11:14 - added alphanumeric exc and more common exceptions, BAE, USAF, RAF, 3M
'v20.04 2014-03-27 10:18 - bugfix to make e.g. " SA " as company name process better
'v20.03 2014-03-24 14:46 - called by xlU_Text_Capitalise_Phrase
'v20.02 2014-03-24 14:30 - bugfixes where last character is a space, hyphenated words
Dim a As Byte, h As Byte, z As Byte, w As String, bReCap As Boolean
If s = "" Then Exit Function
'MIXED CAPITALISE WHOLE STRING (also puts a space at end of string to prevent errors)
    s = UCase(Left(s, 1)) & LCase(Right(s, Len(s) - 1)) & " "
'find first space (word) in string
    a = InStr(s, " ")
'override position for Hyphenated-phrases
    h = InStr(s, "-")
    If h <> 0 And h < a Then a = h
'override position for Badly,spaced,phrases
    h = InStr(s, ",")
    If h <> 0 And h < a Then a = h
'override position for Badly.spaced.phrases
    h = InStr(s, ".")
    If h <> 0 And h < a Then a = h
'override position for (bracketed Phrases)
    h = InStr(s, "(")
    If h <> 0 And h < a Then a = h
'override position for Slash Separated Words/phrases
    h = InStr(s, "/")
    If h <> 0 And h < a Then a = h
'RE-CAPITALISE EXCEPTIONS:
    w = Left(s, a - 1) 'first word in string
'final master exceptions to above logic
    w = xlU_Text_Capitalise_Words_Exceptions(w)
    s = w & Mid(s, a, Len(s))
'rest of phrase
    While a > 0 And a < Len(s)
        z = a + 1
    'MIXED CAPITALISE REST OF STRING (w)
        w = UCase(Mid(s, z, 1)) & Right(s, Len(s) - z)
        s = Left(s, a) & w
    'find next space (word) in string
        a = InStr(z, s, " ")
    'override position for Hyphenated-phrases
        h = InStr(z, s, "-")
        If h <> 0 And (h < a Or a = 0) Then a = h
    'override position for Badly Spaced,phrases
        h = InStr(z, s, ",")
        If h <> 0 And (h < a Or a = 0) Then a = h
    'override position for Badly Spaced.phrases
        h = InStr(z, s, ".")
        If h <> 0 And (h < a Or a = 0) Then a = h
    'override position for (bracketed Phrases)
        h = InStr(z, s, "(")
        If h <> 0 And (h < a Or a = 0) Then a = h
    'override position for (bracketed Phrases)
        h = InStr(z, s, ")")
        If h <> 0 And (h < a Or a = 0) Then a = h
    'override position for Slash Separated Words/phrases
        h = InStr(z, s, "/")
        If h <> 0 And (h < a Or a = 0) Then a = h
    'RE-CAPITALISE EXCEPTIONS:
        w = Mid(s, z, a - z)  'current word
        If Len(w) = 2 Then bReCap = True
        'short words (not many of those!)
        If bReCap = True Then
            s = Left(s, z - 1) & w & Mid(s, a, Len(s))
            bReCap = False  'reset
        End If
    'final master exceptions to above logic
        w = xlU_Text_Capitalise_Words_Exceptions(w)
        s = Left(s, z - 1) & w & Mid(s, a, Len(s))
    Wend
    w = ""
    a = 0
    h = 0
    z = 0
'specify common overrides below
'NB: avoid conflicts with genuine common letter pairs, e.g. SAlvation, UKraine
    s = s & " " 'v2.04 - improves pattern matches by adding space after last char
'common global company extensions
    s = Replace(s, "S R L ", "SRL ")
    s = Replace(s, "L L P ", "LLP ")
    s = Replace(s, "S L U ", "SLU ")
    s = Replace(s, "Sp Zoo ", "Sp Z Oo ")
    s = Replace(s, "Co Kg ", "Co KG ")
'other common exceptions
    s = Replace(s, "Po Box", "PO Box")
    s = Replace(s, "C/o", "C/O")
    s = Replace(s, "T/a", "T/A")
    s = Replace(s, "DE LA ", "De La ")
    s = Replace(s, "-EN-", "-en-")  'Chapel-en-le-Frith
    s = Replace(s, "-LE-", "-le-")  'Chapel-en-le-Frith
    s = Replace(s, " LE ", " le ")  'Hamble le Rice
    'If Right(s, 4) = " AB " Then s = Left(s, Len(s) - 3) & " AB"
    'If Right(s, 4) = " SA " Then s = Left(s, Len(s) - 3) & " SA"
    xlU_Text_Capitalise_Words = s
End Function

Sub xlU_Remind_User(Optional ByVal BypassMessageBox As Boolean = True, Optional AutoCloseAndSave As Boolean = True)
'v37.02 2020-10-08 16:05 - added sAutomationUser and checks multiple VM user accounts
'v37.00 2019-01-16 14:24 - xlU_Remind_User: copied from DMX
'useful for automatically closing unshared workbooks that need to be edited by multiple people
'will reset 5 minute timer if run without ShowMessageBox (do this on Worksheet_SelectionChange event)
'will close & save if AutoCloseAndSave is NOT False
Const iEveryFewMinutes = 5
If InStr(ThisWorkbook.Name, "(master)") = 0 And InStr(sAutomationUser, Environ("Username")) = 0 Then
On Error Resume Next
    If RunWhen <> 0 Then Application.OnTime RunWhen, "xlU_Remind_User", Schedule:=False  'cancel existing OnTime
On Error GoTo 0
    If AutoCloseAndSave Then
        Application.StatusBar = False
        If InStr(Application.Caption, "read-only") > 0 Then
            ThisWorkbook.Close False
        Else
            ThisWorkbook.Close True
        End If
    End If
    If BypassMessageBox Then
        If InStr(Application.Caption, "read-only") > 0 Then
            Exit Sub
        Else
            Application.StatusBar = "WARNING: " & ThisWorkbook.Name & " will close & save automatically at " & Format(RunWhen, "hh:nn:ss")
        End If
    Else
        MsgBox "Reminder: don't leave report open while you're not using it.  Save and close when done, otherwise we can't update it overnight.", vbInformation, ""
        Application.StatusBar = "WARNING: " & ThisWorkbook.Name & " will close & save automatically at " & Format(RunWhen, "hh:nn:ss")
    End If
    RunWhen = Now + TimeSerial(0, iEveryFewMinutes, 0)
    Application.OnTime RunWhen, "xlU_Remind_User", Schedule:=True
End If
End Sub

Sub xlU_Remove_Read_Only()
'v37.02 2020-10-08 16:05 - added sAutomationUser and checks multiple VM user accounts
'v32.03 2017-01-20 12:04 - bugfix to fix file access delays
'v32.01 2017-01-05 15:55 - allows automation on VM
'v32.00 2017-01-05 10:44 - added
On Error GoTo ErrorHandler
Dim bBypassMsgBox As Boolean, xl As Excel.Application
If Application.Name = "Microsoft Excel" Then Set xl = Application Else Exit Sub
If InStr(sAutomationUser, Environ("Username")) > 0 Then bBypassMsgBox = True
If InStr(Application.Caption, "read-only") = 0 Then
    If bBypassMsgBox = False Then MsgBox "Already writeable", vbInformation, ""
Else
    SetAttr ThisWorkbook.FullName, vbNormal

Sub xlU_Save_Make_Read_Only_And_Close()
'v37.02 2020-10-08 16:05 - added sAutomationUser and checks multiple VM user accounts
'v32.06 2017-03-06 10:32 - change to msgbox, easier to understand what's going on
'v32.02 2017-01-20 11:36 - bugfix when saving to temp, kill first
'v32.01 2017-01-05 15:55 - allows automation on VM
'v32.00 2017-01-05 10:44 - added
Dim bBypassMsgBox As Boolean
If InStr(sAutomationUser, Environ("Username")) > 0 Then bBypassMsgBox = True

End If
Dim sThisPath As String

On Error GoTo ErrorHandler

    If bBypassMsgBox = False Then MsgBox "Already read-only", vbInformation, ""
ElseIf Right(sThisPath, 5) <> ".xlsb" Then

    ThisWorkbook.SaveAs Environ("Temp") & "\test.xlsb"
    SetAttr sThisPath, vbReadOnly
    ThisWorkbook.Close False
End If
Exit Sub
ErrorHandler:
    If bBypassMsgBox = False Then MsgBox "Couldn't make writeable, try manually, check for file locks", vbExclamation, ""
End Sub
