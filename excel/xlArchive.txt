Sub do_Archive()
'Call this from modExtra
'add your custom code here to be run during publish routine

If InStr(ThisWorkbook.FullName, "\Testing\") > 0 Then
    Exit Sub

ElseIf InStr(ThisWorkbook.Name, "(NA Domestic)") > 0 Then
    Exit Sub

ElseIf InStr(ThisWorkbook.Name, "Trading Summary by Day") > 0 _
    Or InStr(ThisWorkbook.Name, "Trading Summary by Source") > 0 _
    Or InStr(ThisWorkbook.Name, "Trading Summary by Supplier") > 0 _
    Or InStr(ThisWorkbook.Name, "Trading Summary WOW") > 0 _
    Then
    Call TS_Create_Weekly_Backup

ElseIf InStr(ThisWorkbook.Name, "Trading Summary by Week") > 0 Then
    Call TS_Create_4Weekly_Backup

ElseIf InStr(ThisWorkbook.Name, "Pickups By Day") > 0 Then
    Call PUByDay_Create_Weekly_Backup

ElseIf InStr(ThisWorkbook.Name, "Rate Exception") > 0 Then
    Call REx_Create_Daily_Backup

ElseIf InStr(ThisWorkbook.Name, "PandC_Demand vs ") > 0 Then
    Call Demand_Create_Daily_Backup

ElseIf InStr(ThisWorkbook.Name, "Commercial Setups (nightly)") > 0 Then
    Call Setups_Create_Daily_Backup

End If

End Sub

Private Sub xlArchive_Backup(ByVal pthPub As String, ByVal pthBkp As String, ByVal sReportName As String)
'v3.07 2020-11-06 13:35 - xlArchive_Backup: removed email constants, refers to modEmail now
'v3.06 2020-10-22 15:34 - added Setups_Create_Daily_Backup, set absolute M drive path, added sReportName parameter
'generic routine called from Subs below
On Error GoTo ReportFail
If Dir(pthBkp) <> "" Then
    SetAttr pthBkp, vbNormal
    Kill pthBkp
End If
FileCopy pthPub, pthBkp
SetAttr pthBkp, vbReadOnly
Exit Sub

ReportFail:
modEmail.SendEmail modEmail.cTriggerEmailTo, modEmail.cAdminEmailTo, , sReportName & ": archive failed", "<a href=" & Chr(34) & pthBkp & Chr(34) & ">" & pthBkp & "</a>", True, , , , , True
End Sub

Private Sub TS_Create_Weekly_Backup()
'makes a copy of the report for each week (only runs Monday)

If Weekday(Now, vbMonday) > 1 Then Exit Sub

Dim sDate As String
sDate = Format(Now + 1 - Weekday(Now, vbMonday), "yyyy-mm-dd")

Dim pthPub As String, pthBkp As String, sReportName As String
sReportName = "Trading Summary"
pthPub = admin.Range("admin_OutputFile")
pthBkp = Replace(pthPub, "1 Trading Meetings\", "1 Trading Meetings\Trading Summary Archive\")
pthBkp = Replace(pthBkp, ".xlsb", " (" & sDate & ").xlsb")

Call xlArchive_Backup(pthPub, pthBkp, sReportName)

End Sub

Private Sub TS_Create_4Weekly_Backup()
'makes a copy of the L4W file for each 4 week period (runs every 4th Monday, e.g. week 1, 5, 9, 13, etc.)

If Weekday(Now, vbMonday) > 1 Then Exit Sub

Const cWeekNums As String = "|1|5|9|13|17|21|25|29|33|37|41|45|49|53|"  'in which weeks we publish L4W (Monday)

Dim sWeek As String, sYear As String
sWeek = Application.WeekNum(Now, 21)
If InStr(cWeekNums, "|" & sWeek & "|") = 0 Then Exit Sub
sYear = Year(Now)

Dim pthPub As String, pthBkp As String, sReportName As String
sReportName = "Trading Summary (4wk)"
pthPub = admin.Range("admin_OutputFile")
pthBkp = Replace(pthPub, "1 Trading Meetings\", "1 Trading Meetings\Trading Summary Archive\")
pthBkp = Replace(pthBkp, ".xlsb", " (" & sYear & " Wk " & sWeek & ").xlsb")

Call xlArchive_Backup(pthPub, pthBkp, sReportName)

End Sub
