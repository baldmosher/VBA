'updates comments etc. to the server for all xMX reports
'runs on request from user submissions, or nightly from the daily report, prior to refreshing query
'v6.12 2020-03-18 21:40 - runs modExtra.do_extras_first before submitting, to clear any potential cell/SQL formatting errors
'v6.01 2020-02-10 10:55 - TP: added Rahul's new DMX VBA code to Insert_Audit_Table

Sub Insert_Audit_Table_xMX(ByVal xMX As String, ByVal sqlAuditTable As String, ByVal sUser As String, ByVal DoDebug As Boolean _
    , Optional ByVal dSubmitted As Double)
'allows users to send comments etc. directly to SQL server
'ODBC connection and writer access required on Compliance DB, see https://confluence.bookinggo.io/pages/viewpage.action?pageId=70093636
'admin can also run Insert_Audit_Table "username" to log
'DoDebug = faster if False, if True can check code & timing in Immediate pane
'v6.14 2020-03-24 14:29 - msgbox changed on error and bugfix in uSQL date formatting
'v6.13 2020-03-19 14:00 - msgbox changed on error
'v6.11 2020-03-18 16:11 - minor code cleanup
'v6.05 2020-02-21 15:56 - removed refresh after submit, and Ends when there's an error ('!! this needs to be more user friendly as it still asks to close the file)
'v6.01 2020-02-10 10:55 - TP: added Rahul's new DMX VBA code to Insert_Audit_Table
'v6.00 2020-02-10 10:43 - copied from Update_Audit_Table_xMX (old DMX/IMX routine)

'!! NOTE: ALL DATE FORMATS MUST BE CONVERTED TO "yyyy-mm-dd hh:mm:ss" in uSQL

Dim sErrorLog As String, lErrorLog As Long

Dim sTblRange As String
sTblRange = "Table_" & xMX
Dim sSheet As Worksheet
Set sSheet = Sheets(xMX)

Dim t0 As Double, t1 As Double, t2 As Double, i As Long

t0 = Now

On Error GoTo CannotConnect
Dim cnnstr As String, cnn As Object  'requires reference for Microsoft ActiveX Data Objects Library v2.0 or later to be enabled
Dim uSQL As String
Set cnn = CreateObject("ADODB.Connection")
cnnstr = "Provider=SQLOLEDB;" & _
            "Data Source=bi-01; " & _
            "Initial Catalog=" & sComplianceUploadSchema & ";" & _
            "Trusted_Connection=Yes"
cnn.CommandTimeout = 30
cnn.Open cnnstr
On Error GoTo 0

t1 = Now
If DoDebug Then Debug.Print "Opened connection: " & Format(t1 - t0, "hh:nn:ss")

Dim sReason As String, sPriority As String, sComments As String, sAuthorised_Date As String, sUsername As String, iException As Long, sExcreat As String, iReservation As Long

Dim rTable As Range, r As Long, rmax As Long
Set rTable = Range(sTblRange)
rmax = rTable.Rows.Count

For r = 1 To rmax
    With sSheet
        If Range(sTblRange & "[Reservation]").Rows(r) = "" Then Exit For
        If IsNumeric(Range(sTblRange & "[Reservation]").Rows(r)) = False Then Exit For
        iReservation = .Range(sTblRange & "[Reservation]").Rows(r)
        iException = .Range(sTblRange & "[Exception_ID]").Rows(r)
        sExcreat = .Range(sTblRange & "[Exception_Creation_Date]").Rows(r)
        sReason = Replace(.Range(sTblRange & "[Reason]").Rows(r), "'", "")
        sComments = Replace(.Range(sTblRange & "[Comments]").Rows(r), "'", "")
        sUsername = Replace(.Range(sTblRange & "[Username]").Rows(r), "'", "")
        sAuthorised_Date = Replace(Format(dSubmitted, "yyyy-mm-dd hh:nn:ss"), "'", "")
    End With
    
'!! NOTE: ALL DATE FORMATS MUST BE CONVERTED TO "yyyy-mm-dd hh:mm:ss" in uSQL
    uSQL = "INSERT INTO " & sqlAuditTable & " " & vbLf & _
            "  ([Exception_ID], [Reservation], [Exception_Creation_Date], [Reason], [Comments], [UserName], [Authorised_Date], [Creation_Date], [Modification_Date], [HK_Audit_ID])" & vbLf & _
            "VALUES (" & iException & ", " & iReservation & ", '" & sExcreat & "', '" & sReason & "', '" & sComments & "', '" & sUsername & "', '" & sAuthorised_Date & "', '" & Format(Now, "yyyy-mm-dd hh:nn:ss") & "', '" & Format(Now, "yyyy-mm-dd hh:nn:ss") & "', 0);"
    If DoDebug Then Debug.Print (uSQL)
'only updates when Reason is changed from default AND Username matches submission
    If sReason <> "select" And sUser = sUsername Then
On Error GoTo CompileErrorLog
        cnn.Execute uSQL
On Error GoTo 0
        t2 = Now
        If DoDebug Then Debug.Print "Updated " & r & " rows: " & Format(t2 - t1, "hh:nn:ss")
        i = i + 1
    End If
    t1 = Now
Next r

Debug.Print "Updated " & i & " of " & r - 1 & " rows: " & Format(t2 - t0, "hh:nn:ss")
cnn.Close
Set cnn = Nothing

If sErrorLog <> "" Then
'send failures back to the submitter
    MsgBox "Owing to an error, the following " & lErrorLog & " responses were not logged.  Please close and save the file.  It will now be sent as an attachment to P&C Insight to be loaded manually.  They will confirm when it's been loaded.", vbExclamation, ""
    ThisWorkbook.Save
    modEmail.SendEmail sAdminEmails, sUser, , "Daily Margin Exceptions: responses not logged", "Owing to an error, the attached " & lErrorLog & " responses were not logged.  Please log this manually from the attachment.  If this issue persists, please contact [redacted] for support." & vbLf & vbLf & sErrorLog, , ThisWorkbook.FullName, , , , , , , True
    'modEmail.SendEmail sUser, , , "Daily Margin Exceptions: not all responses logged", "Owing to an error, the following " & lErrorLog & " responses were not logged.  Please try again.  If this issue persists, please contact P&C Insight." & vbLf & vbLf & sErrorLog, , , , , , , , , True
End If

Dim bAEE As Boolean
bAEE = Application.EnableEvents
'Application.EnableEvents = False
'If InStr(Application.Caption, "read-only") = 0 Then xlBackgroundRefresh.Background_Refresh_And_Wait True
'Application.EnableEvents = True

Exit Sub
CompileErrorLog:
    Debug.Print uSQL
    If sErrorLog = "" Then
        sErrorLog = "" & iException
    Else
        sErrorLog = sErrorLog & vbLf & iException
    End If
    lErrorLog = lErrorLog + 1
    Resume Next
CannotConnect:
    MsgBox "Cannot connect to database - please contact P&C Insight for support", vbCritical, ""
End Sub

Sub Fix_Table_Cell_Validation()
'this will fix the cell validation in specified columns
    
Const xmax As Byte = 1
Dim xMX(1 To 2) As String, x As Byte
xMX(1) = "DMX"
xMX(2) = "IMX"
For x = 1 To xmax
    With Sheets(xMX(x)).Range("Table_" & xMX(x) & "[Reason]")
        With .Validation
            .Delete
            .Add Type:=xlValidateList, AlertStyle:=xlValidAlertStop, Operator:= _
            xlBetween, Formula1:="=val_Exceptions_" & xMX(x)
            .IgnoreBlank = True
            .InCellDropdown = True
            .InputTitle = ""
            .ErrorTitle = ""
            .InputMessage = ""
            .ErrorMessage = ""
            .ShowInput = True
            .ShowError = True
        End With
    End With  'Range("Table_" & xmx(x) & "[Reason]")
Next x
End Sub

Function fn_Check_xMX_Validity(ByVal xMX As String) As Boolean
'v3.00 2019-02-25 11:28 - added IMX functions

'v2.05 2017-04-13 13:53 - added fn_Check_DMX_Validity to check report before submitting

Dim rval As Range

        v = Application.Match(r.EntireRow.Cells(1), rHdr, 0)
        If IsError(v) Then MsgBox r.EntireRow.Cells(1)
    End If
Next r
fn_Check_xMX_Validity = True
End Function

Function fn_Check_xMX_Comments(ByVal xMX As String) As Boolean
'v6.10 2020-03-18 12:40 - bugfix in close routine, added dropout clause if no comments found / no changes
'v6.04 2020-02-21 14:21 - bugfix in ranges (r.Row-1)
'v6.03 2020-02-12 16:11 - added validity check for mandatory comments (e.g. IT Issue requires Jira ticket ref)
'v3.05 2019-07-17 09:02 - added fn_Check_DMX_Comments: checks if any comments by user exist before submitting

Dim rReason As Range, rComment As Range
Dim rUser As Range, r As Range, u As String

'1. check username column to see if they actually submitted anything
Set rUser = Sheets(xMX).Range("Table_DMX[Username]")
On Error Resume Next
u = Environ("Username")
For Each r In rUser.Cells
    If r = u Then GoTo CommentFound
Next r
Exit Function

CommentFound:
'2. check Comments exist for Reasons where a Comment is mandatory
Set rReason = Sheets(xMX).Range("Table_DMX[Reason]")
Set rComment = Sheets(xMX).Range("Table_DMX[Comments]")
On Error Resume Next
u = Environ("Username")
For Each r In rUser.Cells
    If r = u Then
        If rReason.Cells(r.Row - 1) = "IT Issue" _
        Or rReason.Cells(r.Row - 1) = "Other" _
        Or rReason.Cells(r.Row - 1) = "Agent Error" Then
            If rComment.Cells(r.Row - 1) = "" Then
                MsgBox "Warning: cannot submit blank comment for Agent Error, IT Issue, or Other.  Please add relevant comment.", vbExclamation, ""
                Application.GoTo rComment.Cells(r.Row - 1)
                fn_Check_xMX_Comments = False
                Exit Function  '!! note: this does NOT cancel the process of closing the workbook
            End If
        End If
    End If
Next r



fn_Check_xMX_Comments = True
End Function
