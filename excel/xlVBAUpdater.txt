Function ReplaceModulesInWorkbook(ByRef wbTarget As Workbook _
    , ByRef ModulePath As String, ByRef ModuleCount As Byte _
    , ByRef iFileRowNumber As Integer, ByRef rFileList As Range) As Boolean
'v3.27 2017-08-03 09:58 - better control of checking version numbers, better on screen messages
'v3.26 2017-08-02 17:00 - bugfix, direct reference to rFileList cells for versions
'v3.25 2017-08-02 16:47 - bugfix, if master version differs from recorded list, it updates the module
'v3.20 2017-07-05 16:43 - no longer activates, seems to cause issues with some workbooks
'v3.15 2017-06-22 12:12 - always deletes specified modules
'v3.08 2017-05-05 14:43 - errorhandlers for set oModule
'v3.07 2017-05-05 11:15 - added safety net for missing mandatory modules (in case deleted in error and master was saved)
'v3.06 2017-05-04 16:21 - renamed uos to cUpd and rejigged offsets
'v3.05 2017-05-04 14:31 - renamed boolean
'v3.03 2017-02-24 18:50 - adds mandatory modules if missing
'v3.01 2017-02-03 11:59 - used uos as Const to fix offsets, only updates module if it doesn't match (ideally want to switch this to earlier in the process to avoid opening files unnecessarily)
'v2.01 2017-01-16 14:07 - adds comment on failure, deletes comment on success
'v1.04 2016-09-16 13:32 - only add admin sheets if modAuto is in the workbook
'v1.03 2016-08-30 13:11 - check if mandatory admin sheets are present
'v1.00 2016-05-03 17:30 - first version

Dim m As Byte, f As Integer, oModule As Object

'this controls later on whether we should copy the admin sheets
Dim bAutomated As Boolean
On Error Resume Next
Set oModule = wbTarget.VBProject.VBComponents("modAuto")
On Error GoTo 0
If Not oModule Is Nothing Then bAutomated = True

Dim bIsMissing As Boolean
Dim sOldVer As String, sNewVer As String

bReplacedModule = False 'reset Private switch

Dim r As Integer
Dim bDeleteIfPresent As Boolean, rDelete As Range
Set rDelete = updVBA_Settings.Range("admin_AlwaysDelete")
Set rDelete = updVBA_Settings.Range(rDelete, rDelete.End(xlDown))
Dim bAddIfMissing As Boolean, rMandatory As Range
Set rMandatory = updVBA_Settings.Range("admin_Mandatory")
Set rMandatory = updVBA_Settings.Range(rMandatory, rMandatory.End(xlDown))
Dim bReplaceIfMismatched As Boolean

f = iFileRowNumber
For m = 1 To ModuleCount
    GenericModuleVersion(m) = rFileList.Cells(1).EntireRow.Cells(cMod + m - 1)
    Application.Goto rFileList.Cells(f)
    If bAutomated Then
    'is this a module to always delete if it's present?
        bDeleteIfPresent = False
        r = 0
        On Error Resume Next
        r = Application.Match(ModuleName(m), rDelete, 0)
        On Error GoTo 0
        If r > 0 Then bDeleteIfPresent = True
        r = 0
    'is this a mandatory module to always add if it's missing?
        bAddIfMissing = False
        r = 0
        On Error Resume Next
        r = Application.Match(ModuleName(m), rMandatory, 0)
        On Error GoTo 0
        If r > 0 Then bAddIfMissing = True
        r = 0
    End If
    If bDeleteIfPresent Or bAddIfMissing Or Left(ModuleName(m), 5) = "admin" Then
        bIsMissing = False
        On Error Resume Next
        Set oModule = Nothing
        Set oModule = wbTarget.VBProject.VBComponents(ModuleName(m))
        On Error GoTo 0
        If oModule Is Nothing Then bIsMissing = True
    End If
    'except when updating the Generic file (f=1), compare version in THIS row with version in Generic file, before updating
    'this saves time re-importing modules unnecessarily
    'requires that Generic file always has the latest version
    ReportModuleVersion(m) = rFileList.Cells(f).EntireRow.Cells(cMod + m - 1)
On Error Resume Next
    bReplaceIfMismatched = False
    sOldVer = ""
    sOldVer = Replace(Application.Run("'" & wbTarget.FullName & "'!" & ModuleName(m) & "." & ModuleName(m) & "Ver"), ModuleName(m) & " ", "")
    Debug.Print "     Checking: " & ModuleName(m) & " " & sOldVer
    If sOldVer <> "" And ReportModuleVersion(m) <> sOldVer Then
        bReplaceIfMismatched = True
        ReportModuleVersion(m) = sOldVer
    End If
    If f = 1 _
    Or ReportModuleVersion(m) <> GenericModuleVersion(m) _
    Or bIsMissing Or bReplaceIfMismatched Then
        sNewVer = ""
        sNewVer = ReplaceOneModule(ModulePath, ModuleName(m), wbTarget, bAutomated, bAddIfMissing, bDeleteIfPresent)
        If sNewVer <> ReportModuleVersion(m) Then
            ReportModuleVersion(m) = sNewVer
            Debug.Print "     Removed : " & ModuleName(m)
            bReplacedModule = True    'if any modules were updated, this triggers save
            If ThisWorkbook.Name = "Generic Sheets for Automation.xlsb" Then
                rFileList.Cells(f).EntireRow.Cells(cMod + m - 1) = ReportModuleVersion(m)
                If rFileList.Cells(f).EntireRow.Cells(cMod + m - 1) <> ReportModuleVersion(m) Then
                    Debug.Print "     FAILED  : " & ModuleName(m)
                End If
            End If
        ElseIf sNewVer <> "" And sOldVer <> sNewVer Then
            Debug.Print "     Updated : " & ModuleName(m) & " " & sNewVer
            ReportModuleVersion(m) = sNewVer
            bReplacedModule = True    'if any modules were updated, this triggers save
            If ThisWorkbook.Name = "Generic Sheets for Automation.xlsb" Then rFileList.Cells(f).EntireRow.Cells(cMod + m - 1) = ReportModuleVersion(m)
        End If
    End If
On Error GoTo 0
Next m

'record the current file module versions, if we are safe to save & close it
If bReplacedModule Then
    For m = 1 To ModuleCount
        If ReportModuleVersion(m) = "" Then
        'module not in workbook
            rFileList.Cells(f).EntireRow.Cells(cMod + m - 1).ClearContents
        Else
        'Record the version number (or X)
            rFileList.Cells(f).EntireRow.Cells(cMod + m - 1).ClearContents
            rFileList.Cells(f).EntireRow.Cells(cMod + m - 1).Value = ReportModuleVersion(m)
        End If
    Next m
'delete the comment from the filename if it's processed OK
    On Error Resume Next
    rFileList.Cells(f).Comment.Delete
    On Error GoTo 0
End If

ReplaceModulesInWorkbook = bReplacedModule
End Function

Function ReplaceOneModule(ByVal ModulePath As String, ByVal ModuleName As String _
    , Optional wbTarget As Workbook, Optional bAutomated As Boolean _
    , Optional bAddModuleIfMissing As Boolean, Optional bDeleteModuleIfPresent As Boolean) As String
'v3.24 2017-08-02 10:34 - bugfixes around adding missing sheets, better error handling
'v3.22 2017-07-31 10:09 - bugfix when copying sheets with existing range names, Application.DisplayAlerts = False
'v3.15 2017-06-22 12:25 - added bDeleteModuleIfPresent exception
'v3.08 2017-05-05 14:43 - added bAlwaysX test
'v3.04 2017-05-02 16:55 - bugfix in version number, prevents failure when _module is still there
'v3.03 2017-02-24 18:50 - bugfix in missing version number when adding new module
'v3.02 2017-02-03 13:09 - commented errorhandler for removing _OLD module.  This does work but only after the macro ends.
'v3.00 2017-01-25 11:31 - added code from http://stackoverflow.com/questions/19800184/vbcomponents-remove-doesnt-always-remove-module
'v2.02 2017-01-23 16:57 - bugfix for NA Domestic reports
'v2.00 2016-11-01 12:13 - rebuild to allow easier self-updates

If wbTarget Is Nothing Then Set wbTarget = ThisWorkbook
If ModuleName = "" Then Exit Function

Dim oModule As Object, wsSource As Worksheet

'force removal of _OLD module, if it's still there from last update
On Error Resume Next
    Set oModule = Nothing
    Set oModule = wbTarget.VBProject.VBComponents(ModuleName & "_OLD")
On Error GoTo 0
    If Not oModule Is Nothing Then
        wbTarget.VBProject.VBComponents.Remove oModule
    End If

'attempt to set module, to see if it exists already
On Error Resume Next
    Set oModule = Nothing
    Set oModule = wbTarget.VBProject.VBComponents(ModuleName)
On Error GoTo 0

Dim rAlwaysX As Range, bAlwaysX As Boolean, bx As Byte
With updVBA_Settings
    Set rAlwaysX = .Range("admin_AlwaysX")
    Set rAlwaysX = .Range(rAlwaysX.Offset(1), rAlwaysX.End(xlDown))
    bx = 0
On Error Resume Next
    bx = Application.Match(ModuleName, rAlwaysX, 0)
On Error GoTo 0
    If bx > 0 Then bAlwaysX = True
End With

If Not oModule Is Nothing Then
'this module/sheet is already present...
    If bAlwaysX Or Left(ModuleName, 5) = "admin" Then
    '...NEVER remove/add this module/sheet, just mark it as present
        ReplaceOneModule = "X"    'default entry
    Else
'remove & re-add module
    On Error Resume Next
        With wbTarget.VBProject.VBComponents
            If oModule.Type = 1 Then    'code module as opposed to sheet/workbook item
                oModule.Name = ModuleName & "_OLD"
                .Remove .Item(ModuleName & "_OLD")
            Else
                .Remove .Item(ModuleName)
            End If
        End With
    On Error GoTo 0
    'test - has it deleted?
        Set oModule = Nothing
    On Error Resume Next
        Set oModule = wbTarget.VBProject.VBComponents(ModuleName)
    On Error GoTo 0
        If oModule Is Nothing Then
        'it's deleted/renamed, so now we can re-import the module
            If bDeleteModuleIfPresent = False Then
                If Dir(ModulePath & ModuleName & ".bas") = "" Then
                'couldn't import the specified module - shared drive access failure or file doesn't exist
                    MsgBox wbTarget.Name & " couldn't access " & ModuleName & ", will now exit. Update it manually or exclude from list by setting future date.  Previous successful updates have been logged." & vbLf & vbLf & ModulePath & ModuleName & ".bas", vbCritical, ""
                    End
                End If
                wbTarget.VBProject.VBComponents.Import ModulePath & ModuleName & ".bas"
            On Error Resume Next
                Set oModule = wbTarget.VBProject.VBComponents(ModuleName)
            On Error GoTo 0
                If Not oModule Is Nothing Then
                'Can we get the version number from the module?
                    ReplaceOneModule = "X"    'default entry
                On Error Resume Next
                    ReplaceOneModule = Replace(Application.Run("'" & wbTarget.FullName & "'!" & ModuleName & "." & ModuleName & "Ver"), ModuleName & " ", "")     'actual version number
                On Error GoTo 0
                'Set the save switch to True, because at least one module has been replaced
                    bReplacedModule = True
                Else
                'couldn't import the module
                    MsgBox wbTarget.Name & " failed to import " & ModuleName & ", will now exit. Update it manually or exclude from list by setting future date.  Previous successful updates have been logged.", vbCritical, ""
                    End
                End If
            End If
        'attempt removal of _OLD module, if it's still there (usually clears it when saving if this doesn't work)
            Set oModule = Nothing
        On Error Resume Next
            Set oModule = wbTarget.VBProject.VBComponents(ModuleName & "_OLD")
        On Error GoTo 0
            If Not oModule Is Nothing Then
                wbTarget.VBProject.VBComponents.Remove oModule  'just try to force it one more time
                If Not oModule Is Nothing Then
                'it hasn't deleted the module - don't add it again and don't log it, just quit
                'commented this because, although it persists, it usually does actually delete the _OLD module after the macro ends
                    'MsgBox wbTarget.Name & " failed to delete " & ModuleName & "_OLD" & ", will now exit. Update it manually or exclude from list by setting future date.  Previous successful updates have been logged.", vbCritical, ""
                    'End
                End If
            End If
        Else
        'it hasn't deleted the module - don't add it again and don't log it, just quit
            MsgBox wbTarget.Name & " failed to delete " & ModuleName & ", will now exit. Update it manually or exclude from list by setting future date.  Previous successful updates have been logged.", vbCritical, ""
            End
        End If
    End If
Else
'this module/sheet is NOT present...
'so do we need to add it?
    If ModuleName = "modExtra" Or bAddModuleIfMissing Then
    'always try to add this module if absent - but NEVER remove it
    On Error Resume Next
        wbTarget.VBProject.VBComponents.Import ModulePath & ModuleName & ".bas"
    'Can we get the version number from the added module?
        ReplaceOneModule = "X"    'default entry
        ReplaceOneModule = Replace(Application.Run("'" & wbTarget.FullName & "'!" & ModuleName & "Ver"), ModuleName & " ", "")     'actual version number
    'Set the save switch to True
        Set oModule = Nothing
        Set oModule = ThisWorkbook.VBProject.VBComponents(ModuleName)
        If Not oModule Is Nothing Then bReplacedModule = True
    On Error GoTo 0
    ElseIf (bAutomated And ModuleName = "pivValidate") _
        Or (bAutomated And Left(ModuleName, 5) = "admin") _
    Then
    'always copy this sheet from Generic Sheets for Automation if missing
    On Error Resume Next
        Set oModule = Nothing
        Set oModule = ThisWorkbook.VBProject.VBComponents(ModuleName)
        If Not oModule Is Nothing Then
            Set wsSource = Nothing
            Set wsSource = ThisWorkbook.Sheets(oModule.Properties("Name").Value)
Application.DisplayAlerts = False
            wsSource.Copy After:=wbTarget.Sheets(wbTarget.Sheets.Count)
Application.DisplayAlerts = True
            Set wsSource = Nothing
            Set wsSource = wbTarget.Sheets(oModule.Properties("Name").Value)
            If Not wsSource Is Nothing Then
                ReplaceOneModule = "X"          'default entry
                bReplacedModule = True
            Else
                ReplaceOneModule = ""           'failed/missing
            End If
        End If
    On Error GoTo 0
    Else
    'Module not in file - clear marker from checklist before re-importing
        ReplaceOneModule = ""
    End If
End If
'just make sure to remove any modModule1, modModule2, etc.
On Error Resume Next
Set oModule = Nothing
Set oModule = wbTarget.VBProject.VBComponents(ModuleName & 1)
If Not oModule Is Nothing Then wbTarget.VBProject.VBComponents.Remove oModule
Set oModule = Nothing
Set oModule = wbTarget.VBProject.VBComponents(ModuleName & 2)
If Not oModule Is Nothing Then wbTarget.VBProject.VBComponents.Remove oModule
Set oModule = Nothing
Set oModule = wbTarget.VBProject.VBComponents(ModuleName & 3)
If Not oModule Is Nothing Then wbTarget.VBProject.VBComponents.Remove oModule
Set oModule = Nothing
On Error GoTo 0
End Function

Function fn_Check_ODC(ByVal wbTgt As Workbook) As String
'v7.01 2019-01-08 10:20 - fn_Check_ODC: changed code to new Group Data share
'v6.06 2018-05-18 16:49 - fn_Check_ODC: bugfix, don't refresh as BackgroundQuery
'v6.03 2018-05-18 12:42 - fn_Check_ODC: refreshes .ODC files to ensure they bring in the current connection details
'v4.08 2018-03-27 15:18 - fn_Check_ODC: checks for ODC files in target
Dim cn As Object, sConn As String, sPath As String
fn_Check_ODC = ""
For Each cn In wbTgt.Connections
    sPath = ""
On Error Resume Next
    sPath = cn.OLEDBConnection.SourceConnectionFile
On Error GoTo 0
    If sPath = "" Then GoTo NextCN  'not an OLEDB connection
    sPath = Replace(sPath, "\\company.local\BackOffice\P&C\", "Y:\")
    sPath = Replace(sPath, "\\company.local\Group Data\P&C\", "Y:\")  'v7.01
    sPath = Replace(sPath, "Y:\RT\Automation\", "M:\Automation\")  'v7.01
    If InStr(sPath, ".odc") > 0 Then
        If fn_Check_ODC = "" Then fn_Check_ODC = "Yes"  'there's an ODC in this file
        If Left(cn.Name, 1) <> "@" Then GoTo UnofficialODC
        With cn.OLEDBConnection
            .SourceConnectionFile = sPath
            .AlwaysUseConnectionFile = True
        End With
        cn.RefreshWithRefreshAll = True
        cn.BackgroundQuery = False
On Error GoTo RefreshFailed 'resumes next
        cn.Refresh
On Error GoTo 0
        cn.BackgroundQuery = True
        If cn.OLEDBConnection.SourceConnectionFile <> sPath Then GoTo UpdateFailed
    End If
NextCN:
Next cn
If fn_Check_ODC = "Yes" Then fn_Check_ODC = "OK"    'no failures
Exit Function
UnofficialODC:
    Debug.Print wbTgt.Name & " | " & cn.Name & " is unofficial | " & sPath
    GoTo NextCN     'only run this on "official" ODC files
RefreshFailed:
    Debug.Print wbTgt.Name & " | " & cn.Name & " didn't refresh | " & sPath
    fn_Check_ODC = "Check"
    Resume Next
UpdateFailed:
    Debug.Print wbTgt.Name & " | " & cn.Name & " didn't update | " & sPath
    fn_Check_ODC = "Check"
    Resume Next
End Function

Function fn_Check_MySQL(ByVal wbTgt As Workbook, Optional DoThisWorkbook As Boolean) As String

    Debug.Print "Checking Data Connections for old MySQL username/connection in " & wbTgt.Name

                    If sOld <> sNew Then
                    'replace with new string

                        Debug.Print "New: " & sNew
                        If fn_Check_PP(wbTgt) Then On Error GoTo ErrorHandlerPP
                        cn.OLEDBConnection.Connection = sNew
                        On Error GoTo 0
                        If sNew = cn.OLEDBConnection.Connection Then
                            GoTo ErrorHandlerVM
                        Else
                            GoTo ErrorHandler
                        End If
                    Else
                        GoTo ErrorHandlerVM
                    End If
                End If
            Else
                GoTo NextConnection
            End If
        End If
    'check OLEDB string

        sOld = ""
