'==========================================================================================
'  Details:
'==========================================================================================
'
'Some neat code to background-refresh all Connections in the workbook, and wait until
'they've refreshed, and calculate when done.  Excel will 'freeze' while it's refreshing
'so be patient.
'
'This should work with any pivot tables, QueryTables, ODBC/OLEDB connections, etc.
'Doing it this way will allow Excel to refresh all the background connections
'simultaneously, which is more efficient than doing one at a time.  However, that does
'put extra strain on the server, so treat this with a little caution.  (The queries
'would be run simultaneously if you did it in SQL Server anyway so don't get too
'worried about that.)
'
'The function will return True/False, depending on whether all the queries updated
'successfully or not.  You can override that (see below)
'
'Always test this before putting it into a live report, and always consider how you
'might prevent end users from refreshing the tables themselves by accident.
'
'Connections block will be added automatically if not present, and formatted.
'
'
'To use it, one of the following example commands:
'
'Call Background_Refresh_And_Wait
'Background_Refresh_And_Wait
'If Background_Refresh_And_Wait() Then MsgBox "Worked" Else MsgBox "Didn't work"
'If Background_Refresh_And_Wait(True) Then MsgBox "This is always True regardless of the result, because you specified that"
'
'
'

Function fn_Connections_Block() As Range
'v4.01 2017-06-29 10:02 - minor performance tweaks
'v2.06 2017-02-20 14:55 - creates or sets formatting for Connections block
'checks admin sheet and sets up Connections box

On Error Resume Next
Dim r As Range, sh As Worksheet
Set sh = ThisWorkbook.Sheets("admin")
If sh Is Nothing Then
    MsgBox "FATAL ERROR: admin sheet missing", vbCritical, ""
    End
End If

'look for Update Log block first as reference point for first column in Connections block
'add (or re-add) range names alongside (this code lets us respecify the columns later if needed)
    Set r = sh.Range("admin_UpdatedTime").EntireRow.Cells(cFirstColumnInConnectionsBox)
    If r Is Nothing Then Exit Function
    If r.EntireColumn.ColumnWidth <> 11.22 Then r.EntireColumn.ColumnWidth = 11.22
    ThisWorkbook.Names.Add "admin_ConnIndex", r
    If r.Value <> "Connections:" Then r.Value = "Connections:"
    Set r = r.Offset(0, 1)
    If r.EntireColumn.ColumnWidth <> 36 Then r.EntireColumn.ColumnWidth = 36
    ThisWorkbook.Names.Add "admin_ConnName", r
    If r.Value <> "Connection Name" Then r.Value = "Connection Name"
    Set r = r.Offset(0, 1)
    If r.EntireColumn.ColumnWidth <> 15 Then r.EntireColumn.ColumnWidth = 15
    If r.EntireColumn.NumberFormat <> "yyyy-mm-dd hh:mm" Then r.EntireColumn.NumberFormat = "yyyy-mm-dd hh:mm"
    ThisWorkbook.Names.Add "admin_ConnTime", r
    If r.Value <> "Updated Time" Then r.Value = "Updated Time"
    If r.Offset(0, 1) <> "" Then r.Offset(0, 1).ClearContents
    
'check it's done
    Set r = Nothing
    Set r = sh.Range("admin_ConnIndex")
    If r Is Nothing Then Exit Function
    Set fn_Connections_Block = r
    
'do the relevant formatting
    With sh.Range(r, r.Offset(0, 2))
        '.Select
        If .Interior.ColorIndex <> cConnectionsBlockHeaderRowColorIndex Then .Interior.ColorIndex = cConnectionsBlockHeaderRowColorIndex
        If .Font.Bold <> True Then .Font.Bold = True
    End With

End Function

Function fn_Switch_Database_Server_In_Connections(ByVal sOldServer As String, ByVal sNewServer As String) As Byte
'v6.00 2017-07-24 17:45 - added fn_Switch_Database_Server_In_Connections
    
Dim cn As Object, c As Byte, cmax As Byte, s As String, IsPP As Boolean, pp As String
cmax = ThisWorkbook.Connections.Count
For c = 1 To cmax
    Set cn = ThisWorkbook.Connections(c)
    If cn.Name = "ThisWorkbookDataModel" Then
        IsPP = True
    Else
    On Error Resume Next    'so it doesn't halt on either condition
        s = ""
        'is it OLEDB?
        s = cn.OLEDBConnection.Connection
        If s <> "" Then
        'yes, it's OLEDB
            If InStr(s, "Data Source=" & sOldServer) > 0 Then
                cn.OLEDBConnection.Connection = Replace(cn.OLEDBConnection.Connection, "Data Source=" & sOldServer, "Data Source=" & sNewServer)
                fn_Switch_Database_Server_In_Connections = fn_Switch_Database_Server_In_Connections + 1
            End If
        Else
        'is it ODBC?
            s = cn.ODBCConnection.Connection
            If s <> "" Then
                If InStr(cn.ODBCConnection.Connection, "Data Source=" & sOldServer) > 0 Then
                    cn.ODBCConnection.Connection = Replace(cn.ODBCConnection.Connection, "Data Source=" & sOldServer, "Data Source=" & sNewServer)
                    fn_Switch_Database_Server_In_Connections = fn_Switch_Database_Server_In_Connections + 1
                End If
            End If
        End If
    On Error GoTo 0
    End If
Next c

pp = "Updated " & fn_Switch_Database_Server_In_Connections & " PivotTable/QueryTable connections"
If IsPP Then pp = pp & vbLf & vbLf & "ThisWorkbook contains a PowerPivot Data Model.  You have to update that manually.  Refer to baldmosher™ for advice."
MsgBox pp, vbInformation, ""
End Function

Sub xlB_Remove_Read_Only_And_Save()
'v8.00 2019-11-22 10:15 - added xlB_Remove_Read_Only_And_Save
'allows for quick update of a report that's refreshed using Simple Refresh process
If MsgBox("ThisWorkbook will be saved and made read only again." & vbLf & vbLf & "TAKE CARE!!!", vbOKCancel Or vbExclamation, "") = vbCancel Then Exit Sub
SetAttr ThisWorkbook.FullName, vbNormal
On Error Resume Next
ThisWorkbook.ChangeFileAccess xlReadWrite
On Error GoTo 0
ThisWorkbook.Save
SetAttr ThisWorkbook.FullName, vbReadOnly
ThisWorkbook.ChangeFileAccess xlReadOnly
MsgBox "ThisWorkbook was saved and made read only again"
End Sub
