Function fn_GetReport_AnyPrefix(ByVal sPrefix As String, ByVal sMessageBody As String) As String
'v6.00 2020-04-08 17:46 - fn_GetReport_AnyPrefix: minor change to info, no code change
'v4.01 2018-06-13 13:36 - fn_GetReport_AnyPrefix: adds any prefix type to logging
'looks for  sPrefix[:] abcdefg
'result is  abcdefg
Dim s As String, a As Integer, b As Integer, c As Integer
On Error Resume Next
sPrefix = Replace(sPrefix & ": ", ": : ", ": ") 'adds ": " if not already specified in sPrefix
    a = InStr(sMessageBody, sPrefix)
    If a = 0 Then Exit Function
    b = a + Len(sPrefix)
    c = InStr(b, sMessageBody, vbLf)
    If c = 0 Then c = Len(sMessageBody) + 1
    s = Mid(sMessageBody, b, c - b)
    s = Replace(s, Chr(10), "") 'replaces line break
    s = Replace(s, Chr(13), "") 'replaces paragraph break
    While Right(s, 1) = " "     'removes space from end
        s = Left(s, Len(s) - 1)
    Wend
fn_GetReport_AnyPrefix = s
    'MsgBox fn_GetReport_AnyPrefix & "<end>"
End Function

Function Audit_RegEx_Reservation(ByVal sSubject As String, ByVal sBodyText As String) As Long
'v3.05 2018-05-30 15:36 - Audit_RegEx_Reservation: first version
'https://blogs.technet.microsoft.com/heyscriptingguy/2005/04/15/how-can-i-search-a-text-file-for-9-digit-numbers/

Dim objRegEx As Object
Set objRegEx = CreateObject("VBScript.RegExp")
objRegEx.Global = True
objRegEx.Pattern = "\d{9}"  'finds the first 9 digit number, but it will also find a 10 digit number...

Dim strSearchString As String, strMatch, colMatches

'check the Subject first (more accurate)
strSearchString = sSubject
Set colMatches = objRegEx.Execute(strSearchString)
If colMatches.Count > 0 Then
   For Each strMatch In colMatches
       Audit_RegEx_Reservation = strMatch.Value
       Exit Function
   Next
End If

'check the Body (less accurate)
strSearchString = sBodyText
Set colMatches = objRegEx.Execute(strSearchString)
If colMatches.Count > 0 Then
   For Each strMatch In colMatches
       Audit_RegEx_Reservation = strMatch.Value
       Exit Function
   Next
End If

End Function
