'--- CC-SUMMARY-START ---
'===========================================================================
' modSleep: kernel32 Sleep declaration and file-size delay helper
'
' Provides the Windows kernel32 Sleep API declaration and a helper function
' (zSleepTime) that returns a delay in milliseconds proportional to a file's
' size — useful when waiting for a file write to complete before reading it.
' Paste into modAuto or any module needing Sleep; make the declaration
' Private to avoid conflicts with other modules.
'===========================================================================
'--- CC-SUMMARY-END ---
'paste this into a module (e.g. modAuto) and make Sleep Private not Public
'Syntax:
'   Sleep zSleepTime(pthfn)     (time linked to file size)
'   Sleep zSleepTime            (1000ms)

Public Declare PtrSafe Sub Sleep Lib "kernel32" (ByVal dwMilliseconds As LongPtr)

Private Function zSleepTime(Optional ByVal pthfn As String) As Integer
Dim dDelay As Double
Const iDefault As Long = 1000       '1s delay when no pthfn specified
Const iDivisible As Long = 100000   'sensible divisor to convert bytes into ms delay
If pthfn = "" Or Dir(pthfn) = "" Then
    zSleepTime = iDefault
'ElseIf Dir(pthfn) = "" Then
'    zSleepTime = iDefault
Else
    dDelay = FileLen(pthfn) / iDivisible    'delay in ms based on existing file size
    zSleepTime = dDelay                     'delay in ms based on existing file size
End If
End Function
