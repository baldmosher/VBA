Function fn_replace_report_links(ByVal sOriginalCode As String)
'v13.13 2020-11-05 15:25 - fn_replace_report_links: switched to sPathMDrive from sPathYDrive
'v11.18 2018-03-26 15:02 - added sPathYDrive
'v8.00 2016-04-08 12:35 - replaces "link%20%to%20file" in HTML
'tries to replace all instances of "Y:\Path%20to%20Folder\Link"
'with a formatted HTML tag for "\\company.local\Group Data\P&C\Path to Folder\Link"
'(note: quote marks and %20 spaces are required to prevent errors and reprocessing)
Dim sNewCode As String
sNewCode = sOriginalCode
Const cMtag As String = "M:\"
Const cUNC As String = sPathMDrive
Const cQuo As String = "&quot;"  'in HTML, this is equal to Chr(34) in VBA or " in text
Const cPlaceholder As String = "CORRECTEDLINKGOESHERE"
Dim a As Integer, b As Integer, c As Long, sLink As String, sUNC As String
c = Len(sNewCode)  'end marker, corrected later if applicable

'does the code appear to contain a suitable link?
a = InStr(a + 1, sNewCode, cQuo & cMtag)

While a > 0
    'check that link is "surrounded by quote marks"
    'find next "quote mark" that marks the end of the link
    b = InStr(a + 1, sNewCode, cQuo)
    If b > 0 Then
    'next range to search
        c = b + Len(cQuo)
    'take the entire link with quotemarks
        sLink = Mid(sNewCode, a, b - a + Len(cQuo))
    'check for spaces, this will cause errors
        If InStr(sLink, " ") > 0 Then
            MsgBox "Can't convert link with spaces into HTML, replace spaces with %20", vbCritical, "fn_replace_report_links"
            fn_replace_report_links = sOriginalCode
            Exit Function
        End If
    'replace the entire link with a placeholder in code
        sNewCode = Replace(sNewCode, sLink, cPlaceholder)
    'create UNC from link
        sUNC = Replace(sLink, cMtag, cUNC)  'replace M:\ with \\UNC\
        sUNC = Replace(sUNC, cQuo, Chr(34)) 'replace quotes with " suitable for <a> HTML tag
        sUNC = Replace(sUNC, "%20", " ")    'replace HTML spaces with actual spaces
    'clean up the link
        sLink = Replace(sLink, cQuo, "")
        sLink = Replace(sLink, "%20", " ")
    'replace placeholder with corrected UNC and link in HTML tags
        sNewCode = Replace(sNewCode, cPlaceholder, "<a href=" & sUNC & ">" & sLink & "</a>")
        Debug.Print sLink
        Debug.Print sUNC
        Debug.Print sNewCode
    End If
    a = InStr(c, sNewCode, cQuo & cMtag)
Wend
fn_replace_report_links = sNewCode
End Function

Sub Outlook_Clean_Emails(Optional ByVal AlsoSeparate As Boolean)
'v11.22 2018-04-24 14:26 - added AlsoSeparate
'v11.00 2016-10-20 14:39 - Outlook_Clean_Emails() first version
'takes long string from ActiveCell of "Email Recipient <email@address.com>"
'and extracts just "email@address.com" into cell below
'>> optionally splits emails by cSep and dumps vertically into a column (allows sorting)
Dim sFull As String, sThis As String, sOutput As String
Dim a As Integer, b As Integer, c As Integer
Const cStt As String = "<"
Const cEnd As String = ">"
Const cSep As String = "; "
sFull = ActiveCell
a = 1
NextaClean:
a = InStr(a, sFull, cStt) + 1
If a > 1 Then
    b = InStr(a, sFull, cEnd)
    sThis = Mid(sFull, a, b - a)
    If InStr(sOutput, sThis) = 0 Then
    'removes duplicates
        sOutput = sOutput & sThis & cSep
    End If
    a = b + 1
    GoTo NextaClean
End If
If Right(sOutput, 2) = cSep Then sOutput = Left(sOutput, Len(sOutput) - 2)
If sOutput <> "" Then
    If AlsoSeparate = False Then
        ActiveCell.Offset(1) = sOutput
    Else
        a = 1
        c = 1
NextaSplit:
        b = InStr(a, sOutput, cSep) + 1
        If b > a Then
            sThis = Mid(sOutput, a, b - a - 1)
            ActiveCell.Offset(c) = sThis
            c = c + 1
            a = b + Len(cSep) - 1
            GoTo NextaSplit
        Else
            sThis = Mid(sOutput, a, Len(sOutput))
            ActiveCell.Offset(c) = sThis
        End If
    End If
End If
End Sub
