Sub pivValidate_Filter_Reset()
'v4.23 2016-08-30 11:48 - code tweak, no longer requires pivValidate sheet
'v4.22 2016-08-30 09:50 - requires xlPivots v6.01 or later
'v4.21 2016-08-26 09:01 - calculates after filtering to update "chk_" ranges
'v4.20 2016-07-20 11:34 - sends email if date filters can't be applied
'for each pivot on pivValidate sheet,
'---> resets to 1 any PageFields: L4W, LW, WTD, LWE, L7D, LD, etc.
'---> resets to 0 any PageFields: ConvExcl, GrossExcl
'won't work correctly if a pivot has >1 date filter or exclusion field

On Error Resume Next
Dim pv As Worksheet, pt As PivotTable, pf As PivotField
Set pv = ThisWorkbook.Sheets("pivValidate")
If pv Is Nothing Then Exit Sub
For Each pt In pv.PivotTables
'time period
    Set pf = pt.PageFields("L4W")
    Set pf = pt.PageFields("WTD")
    Set pf = pt.PageFields("LW")
    Set pf = pt.PageFields("LD")
    Set pf = pt.PageFields("LWE")
    Set pf = pt.PageFields("LDLW")
    Set pf = pt.PageFields("PW")
    If Not pf Is Nothing Then
        If pf.CurrentPage <> 1 Then pf.CurrentPage = 1
        If pf.CurrentPage <> 1 Then
            Call do_Send_Filter_Warning(pt.Name, pf.Name)
            Exit Sub
        End If
        Set pf = Nothing
    End If
'conv/gross
    Set pf = pt.PageFields("ConvExcl")
    Set pf = pt.PageFields("GrossExcl")
    If Not pf Is Nothing Then
        If pf.CurrentPage <> 0 Then xlPivots.PageField_Clear pf
        If pf.CurrentPage <> 0 Then pf.CurrentPage = 0
        If pf.CurrentPage <> 0 Then
            Call do_Send_Filter_Warning(pt.Name, pf.Name)
            Exit Sub
        End If
        Set pf = Nothing
    End If
Next pt
Application.Calculate
End Sub

'v6.01 2017-08-02 14:54 - SaveNewTestVersion: code layout, changed to Environ("Username")
'v6.00 2017-08-01 17:19 - added SaveNewTestVersion

Dim bAEE As Boolean
bAEE = Application.EnableEvents

On Error Resume Next

With wbTarget

'1. checks Testing folder for latest test version
    Dim sCurrentTestFile As String, sNewTestFile As String
    Dim sCurrentTestVer As String, sNewTestVer As String
    Dim dCurrentTestVer As Double, dNewTestVer As Double
    Dim v As Integer
    sCurrentTestFile = modFile.Find_Newest_Filename(.Sheets("admin").Range("admin_Working"), Left(.Name, InStr(.Name, " (master)")))
    If sCurrentTestFile = "" Then
        sCurrentTestFile = .Sheets("admin").Range("admin_Working") & Replace(.Name, "(master)", "v1-00 (master)")
    End If
    If sCurrentTestFile = "" Then Exit Function
    v = 0
    v = InStr(sCurrentTestFile, " v")
    While Not IsNumeric(Mid(sCurrentTestFile, v + 2, 1))
        v = InStr(v + 2, sCurrentTestFile, " v") 'avoids errors with " vs " in filename
    Wend
    sCurrentTestVer = Mid(sCurrentTestFile, v + 1, InStr(v + 1, sCurrentTestFile, " ") - v - 1) 'result: v1-00
    sNewTestVer = sCurrentTestVer
'increments minor version by .01 to save new test version
'NOTE: this ignores ChangeLog history, and assumes the latest version in Testing folder is current
    sNewTestVer = Left(sNewTestVer, InStr(sNewTestVer, "-")) & Format(Right(sNewTestVer, 2) + 1, "00")
    With .Sheets("ChangeLog")
    If .Cells(1, 2).Value <> sNewTestVer Then
        .Columns(2).EntireColumn.Copy
        .Columns(2).EntireColumn.Insert
Application.CutCopyMode = False
        .Columns(2).EntireColumn.ClearContents
        .Cells(1, 2).Value = sNewTestVer
    End If
        .Cells(2, 2).Value = Environ("Username")
        .Cells(3, 2).Value = Now
    End With
    
'2. save a new copy of the master file to the Testing folder with the next version number
Dim bOpenNewFile As Boolean, sNewFilePath As String
sNewFilePath = .Sheets("admin").Range("admin_Working") & Replace(sCurrentTestFile, sCurrentTestVer, sNewTestVer)
bOpenNewFile = False

If ThisWorkbook.FullName = .Sheets("admin").Range("admin_Working") & sCurrentTestFile Then
'this is the test version: increment, open & close
    bOpenNewFile = True
    .Sheets("ChangeLog").Cells(4, 2).Value = "<add notes here>"
Else
'create the new test version with increment
    .Sheets("ChangeLog").Cells(4, 2).Value = "auto-updated modules"
End If

Application.EnableEvents = False
    .SaveCopyAs sNewFilePath
Application.EnableEvents = bAEE
If bOpenNewFile Then
    MsgBox "New version " & sNewTestVer & " created.  Please update Changelog", vbInformation, ""
    Sleep 1000
On Error GoTo ErrorCouldNotOpen
    If Dir(sNewFilePath) <> "" Then Workbooks.Open sNewFilePath
On Error GoTo 0
Application.EnableEvents = False
    .Close False    'this will end macros in ThisWorkbook/wbTarget
Application.EnableEvents = bAEE
End If

ErrorCouldNotOpen:
End With 'wbTarget

'3. use this to update the version number in the Files list if running from Generic Sheets
    SaveNewTestVersion = sNewTestVer

End Function

Private Sub admin_Rearrange_Sheets()
'v6.37 2020-10-23 08:37 - admin_Rearrange_Sheets: error handler allows for missing admin sheets
'v6.35 2020-10-08 16:05 - added sAutomationUser and checks multiple VM user accounts
'v6.30 2019-03-22 09:34 - admin_Rearrange_Sheets: deletes pivValidate if it exists
'v6.29 2019-03-22 09:24 - admin_Rearrange_Sheets: only smax = 6 admin sheets, deletes Access Control sheet if exists
'v6.11 2017-09-05 09:50 - admin_Rearrange_Sheets: bugfix for reports with charts
'v6.10 2017-08-23 15:37 - admin_Rearrange_Sheets: first version
'Call this before AND after saving a test version (only does anything if it's needed)
'ALL sheets have to be visible to prevent errors - so make sure they are hidden or shown afterwards in the routine
'arranges the admin sheets into the same order for consistency in all reports

'only do this for test versions
If InStr(ThisWorkbook.Path, "Automation\Testing\") = 0 Then Exit Sub

Dim bAEE As Boolean, iAC As Integer
iAC = Application.Calculation
Application.Calculation = xlCalculationManual
bAEE = Application.EnableEvents
Application.EnableEvents = False

'show all sheets in workbook (otherwise we can't move them)
Dim ws As Worksheet
For Each ws In ThisWorkbook.Worksheets
    If ws.Visible <> xlSheetVisible Then ws.Visible = xlSheetVisible
Next ws

'SHEET ORDER:
Dim sh() As Worksheet, s As Byte, smax As Byte
Dim rSrc As Range, rTgt As Range, sPiv As String
smax = 6
ReDim sh(1 To smax) As Worksheet
With ThisWorkbook
On Error Resume Next                            'only if you find the sheet

'delete admin sheets that are no longer required:
'Access Control
    Set sh(1) = .Sheets("Access Control")       'no longer required after modAuto v6.27
    If Not sh(1) Is Nothing Then
        Application.DisplayAlerts = False
        sh(1).Delete
        Set sh(1) = Nothing
        Application.DisplayAlerts = True
    End If
'pivValidate
'!! see below, eventually, need to consider deleting this sheet completely, but this prepares for it in 90% of cases
GoTo AvoidThisPart
    Set sh(1) = .Sheets("pivValidate")          'no longer required after modAuto v6.30
    If Not sh(1) Is Nothing Then
        s = sh(1).PivotTables.Count
        While s > 0                             'at least 1 pivot needs moving
            With sh(1).PivotTables(1)
                Set rSrc = .TableRange2
                Application.GoTo rSrc
                sPiv = .Name
            End With
            If Not rSrc Is Nothing Then
                Set rTgt = .Sheets("Validation").Cells.SpecialCells(xlCellTypeLastCell).Offset(0, 2).EntireColumn.Cells(20)
                Application.GoTo rTgt
                rSrc.Cut
                .Sheets("Validation").Paste
                With rTgt.Cells(1)
                    .Offset(-3) = sPiv
                    .Offset(-3).Font.Bold = True
                    .Offset(-5) = "* moved from pivValidate *"
                    .Offset(-5).Font.Bold = True
                End With
                sPiv = ""
                Set rSrc = Nothing
                Set rTgt = Nothing
                s = sh(1).PivotTables.Count
            End If
        Wend
    'eventually, need to consider deleting the sheet completely, but this prepares for it in 90% of cases
        'Application.DisplayAlerts = False
        'sh(1).Delete
        'Set sh(1) = Nothing
        'Application.DisplayAlerts = True
    End If
AvoidThisPart:

'mandatory modAuto sheets moved first:
    Set sh(1) = .Sheets("MACROS")               'always sheet 1
    Set sh(2) = .Sheets("admin")                'always sheet 2
    Set sh(3) = .Sheets("Changelog")            'always sheet 3, beyond this, the order is less important
    Set sh(4) = .Sheets("Email Notification")   'not always used by workbook
    Set sh(5) = .Sheets("Validation")           'this is the main sheet for cubesets, etc.
    Set sh(6) = .Sheets("pivValidate")          'only needed for @ThisMonday/@Today as we don't use the validation lookups file
   'smax = ^ (this is set above the ReDim)
    For s = 1 To smax
        If Not sh(s) Is Nothing Then            'sheet exists in workbook
            If sh(s).Index <> s Then
                Debug.Print "Moving " & sh(s).Name & " to index " & s
                sh(s).Move Before:=.Sheets(s)
                If sh(s).Index = s Then
                    Debug.Print "Moved OK"
                Else
                    If InStr(sAutomationUser, Environ("Username")) = 0 Then MsgBox "Sheet move failed: " & sh(s).Name & " not in correct index location (" & s & "), please move it manually"
                End If
            End If
        End If
    Next s
End With
On Error GoTo 0
Application.EnableEvents = bAEE
Application.Calculation = iAC
End Sub

        If r.Cells(1) <> s Then r.Cells(1) = s

        If r.Cells(2) <> s Then r.Cells(2) = s
    s = "All dates calculated relative to TODAY"
        If r.Cells(3) <> s Then r.Cells(3) = s
    'two blank cells below (NB: Offset(1,2) will be "Updated?"
        If r.Cells(3).Offset(1) <> "" Then r.Cells.Offset(1).EntireRow.Insert
        If r.Cells(3).Offset(2) <> "" Then r.Cells.Offset(1).EntireRow.Insert
    End If

'remove row below chk_Data ranges if it says "<< add link"
    Set r = Nothing
    Set r = admin.Range("chk_Sessions").Offset(1, -1)
    If Not r Is Nothing Then
        If Left(r.Text, 11) = "<< add link" Then r.EntireRow.Delete
    End If
    Set r = Nothing
    Set r = admin.Range("chk_Bookings").Offset(1, -1)
    If Not r Is Nothing Then
        If Left(r.Text, 11) = "<< add link" Then r.EntireRow.Delete
    End If
    Set r = Nothing
    Set r = admin.Range("chk_CI").Offset(1, -1)
    If Not r Is Nothing Then
        If Left(r.Text, 11) = "<< add link" Then r.EntireRow.Delete
    End If
    Set r = Nothing
    Set r = admin.Range("chk_Filters").Offset(1, -1)
    If Not r Is Nothing Then
        If Left(r.Text, 11) = "<< add link" Then r.EntireRow.Delete
    End If

'2 blank rows above OUTPUT FILE
    Set r = Nothing
    Set r = admin.Range("admin_OutputFile").Offset(-1)
    If Not r Is Nothing Then
        r.Font.Underline = True     'this isn't formatted correctly in some admin sheets
        While r.Offset(-2) <> ""
            r.EntireRow.Insert
        Wend
    End If

'2 blank rows above "Update Log"
    Set r = Nothing
    Set r = admin.Range("admin_UpdatedTime")
    If Not r Is Nothing Then
        Set r = r.EntireRow.Cells(1).Offset(-1)
        If Not r Is Nothing Then
            If r = "Update Log" Then
                While r.Offset(-2) <> ""
                    r.EntireRow.Insert
                Wend
            End If
        End If
    End If

'1 Placeholder row below "Update Log"
    Set r = Nothing
    Set r = admin.Range("admin_UpdatedTime")
    If Not r Is Nothing Then
        Set r = r.Offset(1)
        If r = 0 And r.Offset(1) <> 0 Then
            admin.Range(r.EntireRow.Cells(1), r.EntireRow.Cells(admin.Range("admin_PublishedTime").Column)).Delete Shift:=xlUp
        End If
    End If

'default email recipients
    Set r = Nothing
    Set r = admin.Range("admin_EmailTO")
    s = sEmailVM
        If r <> s Then r = s
    Set r = Nothing
    Set r = admin.Range("admin_EmailCC")
    s = sEmailTeam
        If r <> s Then r = s
    
Application.ScreenUpdating = bASU
Application.EnableEvents = bAEE
Application.Calculation = iAC

End Sub
