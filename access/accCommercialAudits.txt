Function fn_JIRA_CDate(ByVal sDateString As String) As Date
'v2.00 2019-03-08 11:43 - version numbering started
'Converts JIRA date into Access date when loading in from TXT files
'Note: XLS import can just use standard CDate function
On Error Resume Next
Dim sCleaned As String
Const DoDebug As Boolean = False
If sDateString = "" Then Exit Function
sCleaned = "20" & Mid(sDateString, 8, 2) & "-" & Mid(sDateString, 4, 3) & "-" & Mid(sDateString, 1, 2) & " " & Mid(sDateString, 11, 10)
fn_JIRA_CDate = CDate(sCleaned)
If fn_JIRA_CDate = 0 Then Exit Function
If DoDebug Then Debug.Print sDateString & " >> " & sCleaned & " >> " & fn_JIRA_CDate
End Function

Function fn_JIRA_weekday(ByVal dDate As Date) As String
'v2.00 2019-03-08 11:43 - version numbering started
'finds the relevant Monday of the week
On Error Resume Next
Const DoDebug As Boolean = False
Dim s As String, d As Double, wd As Byte
If dDate = 0 Then Exit Function
    If DoDebug Then Debug.Print dDate
d = dDate
    If DoDebug Then Debug.Print "d:  " & d
wd = Weekday(d, vbMonday)
    If DoDebug Then Debug.Print "wd: " & wd
d = d - wd + 1
    If DoDebug Then Debug.Print "d: " & d
s = Format(d, "yyyy-mm-dd")
    If DoDebug Then Debug.Print "s: " & s
fn_JIRA_weekday = s
End Function

Function jira_Destination(Optional ByRef sDestinationList As String = "", Optional sDestinationText As String = "")
'v6.02 2020-10-06 20:38 - jira_Destination: handles "multiple" countries
'v5.06 2020-04-14 20:31 - added jira_Destination to combine "Destination Country (DN)" from list with "Destination Country" free text
Dim s As String
If sDestinationList = "" Then
    s = sDestinationText
ElseIf sDestinationList = "multiple countries -- specified below" Then
    s = "multiple: " & sDestinationText
Else
    s = sDestinationList
    If Left(Right(s, 5), 2) = " (" And Right(s, 1) = ")" Then s = Left(s, Len(s) - 5)
End If
jira_Destination = s
End Function

Function jira_VendorID(Optional ByRef sVendorID1 As String = "", Optional sVendorID2 As String = "")
'v6.04 2020-10-07 09:38 - jira_VendorID: handles "multiple" Vendor fields and chooses the best one
Dim s As String
If sVendorID1 = "" And sVendorID2 = "" Then
    s = ""
ElseIf sVendorID1 = "" Then
    s = sVendorID2
Else
    s = sVendorID1
End If
jira_VendorID = s
End Function
