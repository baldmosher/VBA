Attribute VB_Name = "accAuto"
Option Compare Database

'--- CC-SUMMARY-START ---
'===========================================================================
' accAuto: AutoExec automation controller (pcinsightvm)
'
' Entry point for scheduled overnight report automation on the pcinsightvm
' Access database. Runs a configured task sequence between 18:00-18:59;
' prompts for manual confirmation at any other time.
' Requires: modEmail v11.12+, WScript reference
'===========================================================================
'--- CC-SUMMARY-END ---
Public Const sAutoUser As String = "pcinsightvm"
Public Const sEmailToAdmin As String = "recipient@company.com; pcinsightvm@company.com"
Public Const sRecipients As String = "p&cinsight@company.com"

'Version History:
Public Const accAutoCodeVer As String = _
"accAuto v3.04"
'v3.04 2020-10-20 17:22 - merged forks for pcopsvm and pcinsightvm, but keep Consts separate
'v3.03 2020-10-15 13:04 - only runs AutoExec at 18:00-18:59hrs, otherwise asks

'v1.00 2017-0?-?? ??:?? - first version

Function accAutoVer() As String
accAutoVer = accAutoCodeVer
End Function

Function AutoExec(ByVal sNameOfMacro, Optional QuitWhenDone = True)
'v3.03 2020-10-15 13:04 - only runs AutoExec at 18:00-18:59hrs, otherwise asks
If CreateObject("WScript.Network").UserName <> sAutoUser Then Exit Function
If Hour(Now) = 18 Then GoTo AutoExec
If MsgBox("Outside normal running time.  Run AutoExec anyway?", vbQuestion Or vbYesNo, "") = vbNo Then Exit Function

AutoExec:
    DoCmd.RunMacro sNameOfMacro
    Send_Email_Success
    If QuitWhenDone Then DoCmd.Quit
End Function

Sub Send_Email_Success()
'v3.04 2020-10-20 17:22 - merged forks for pcopsvm and pcinsightvm, but keep Consts separate
    modEmail.SendEmail sEmailToAdmin, sRecipients, _
        , "Successfully updated: " & CurrentProject.Name _
        , "Access database updated." _
            & "<p>" & "Database: <a href=" & Chr(34) & CurrentProject.FullName & Chr(34) & ">" & CurrentProject.Name & "</a>" _
            & "<p>" & accAuto.accAutoCodeVer & "<br>" & modEmail.modEmailCodeVer _
            , False, , , , , True
End Sub
